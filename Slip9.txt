Q1. Hill Climbing Algorithm (10 Marks)


def f(x):
    return -x*x + 4*x

x = 0
step = 0.1

while True:
    if f(x + step) > f(x):
        x += step
    elif f(x - step) > f(x):
        x -= step
    else:
        break

print("Maximum x:", round(x, 2))
print("Maximum value:", round(f(x), 2))





Q2. Forward Chaining (20 Marks)


facts = {'rain'}
rules = [
    ({'rain'}, 'wet_ground'),
    ({'wet_ground'}, 'slippery'),
    ({'slippery'}, 'drive_carefully')
]

changed = True

while changed:
    changed = False

    for condition, conclusion in rules:
        if condition.issubset(facts) and conclusion not in facts:
            facts.add(conclusion)
            changed = True

print("Facts:", facts)
