Q1. A* Search Algorithm (10 Marks)


import heapq

graph = {
    'A': [('B', 1), ('C', 3)],
    'B': [('D', 2)],
    'C': [('D', 1)],
    'D': []
}

h = {'A': 3, 'B': 2, 'C': 1, 'D': 0}

def astar(start, goal):
    q = [(h[start], 0, start, [start])]
    visited = set()

    while q:
        f, cost, node, path = heapq.heappop(q)

        if node == goal:
            print("Path:", path)
            print("Cost:", cost)
            return

        if node in visited:
            continue
        visited.add(node)

        for n, w in graph[node]:
            if n not in visited:
                g = cost + w
                heapq.heappush(
                    q, (g+h[n], g, n, path+[n])
                )

astar('A', 'D')






Q2. ANN for Linear Regression (20 Marks)



import numpy as np
from sklearn.neural_network import MLPRegressor
from sklearn.metrics import mean_squared_error

X = np.array([[1], [2], [3], [4], [5]])
y = np.array([2, 4, 6, 8, 10])

model = MLPRegressor(
    hidden_layer_sizes=(5,),
    learning_rate_init=0.01,
    max_iter=5000,
    random_state=42
)

model.fit(X, y)
pred = model.predict(X)

print("Predictions:", pred)
print("MSE:", mean_squared_error(y, pred))
